constexpr int MAX=2e5+10;
constexpr int LEN=20;
int sparse[MAX][LEN];

void build(int n) {
    for(int i=0;i<n;i++){
        sparse[i][0]=arr[i];
    }
    for(int j=1;(1<<j)<=n;j++){
        for(int i=0;i+(1<<j)<=n;i++){
            sparse[i][j]=min(
                sparse[i][j-1], sparse[i+(1<<(j-1))][j-1]
            );
        }
    }
}
