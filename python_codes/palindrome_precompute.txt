INF = 10**9
pal = []

for i in range(1, 100000):
    x = i
    t = i // 10

    while t:
        x = x * 10 + t % 10
        t //= 10

    if x <= INF:
        pal.append(x)

    x = i
    t = i

    while t:
        x = x * 10 + t % 10
        t //= 10

    if x <= INF:
        pal.append(x)

pal.sort()
