def build(n, arr):
    for i in range(n):
        sparse[i][0] = arr[i]
    j = 1
    while (1 << j) <= n:
        i = 0
        while i + (1 << j) <= n:
            sparse[i][j] = min(
                sparse[i][j - 1],
                sparse[i + (1 << (j - 1))][j - 1]
            )
            i += 1
        j += 1
