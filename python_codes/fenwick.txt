def get(i):
    i += 1
    ans = 0

    while i > 0:
        ans += fen[i]
        i -= (i & -i)

    return ans


def update(i, val, n):
    i += 1

    while i <= n:
        fen[i] += val
        i += (i & -i)


def query(l, r):
    return get(r) - get(l - 1)


# Usage:
#
# fen = [0] * (n+1)
#
# update(idx, val, n)   # add val at 0-based index idx
#
# pref = get(i)         # sum of [0..i]
#
# sum = query(l, r)     # sum of [l..r]
#
# Notes:
# 1. External indexing is 0-based.
# 2. Pass actual array size as n.
# 3. fen must be at least size n+1.
